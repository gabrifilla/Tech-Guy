// Unity MCP execute_code method body. Replace visual children, Avatar, and visual references only.
if (EditorApplication.isPlaying) throw new System.InvalidOperationException("Exit Play Mode before scene authoring.");
const string modelPath="Assets/_Project/Art/Characters/TG/TG_Rigged.fbx";
var model=AssetDatabase.LoadAssetAtPath<GameObject>(modelPath);
var avatar=AssetDatabase.LoadAllAssetsAtPath(modelPath).OfType<Avatar>().SingleOrDefault();
if (!model || !avatar || !avatar.isHuman || !avatar.isValid) throw new System.InvalidOperationException("A valid imported TG Humanoid is required.");
var paths=new[] { "Assets/_Project/Scenes/NexusLobby.unity", "Assets/_Project/Scenes/FirstSector.unity", "Assets/_Project/Scenes/PrologueTutorial.unity", "Assets/_Project/Scenes/Playground.unity", "Assets/_Project/Scenes/CombatStudy.unity", "Assets/_Project/Scenes/ProceduralStageAuthoring.unity" };
var active=UnityEngine.SceneManagement.SceneManager.GetActiveScene();
var originallyLoaded=Enumerable.Range(0,UnityEngine.SceneManagement.SceneManager.sceneCount).Select(i=>UnityEngine.SceneManagement.SceneManager.GetSceneAt(i)).ToArray();
if (originallyLoaded.Any(s=>s.isDirty)) throw new System.InvalidOperationException("An open scene has unsaved changes. Preserve a live snapshot before running scene integration.");
var results=new System.Collections.Generic.List<object>();
foreach(var path in paths)
{
    var scene=UnityEngine.SceneManagement.SceneManager.GetSceneByPath(path);
    bool opened=!scene.IsValid() || !scene.isLoaded;
    if(opened) scene=UnityEditor.SceneManagement.EditorSceneManager.OpenScene(path,UnityEditor.SceneManagement.OpenSceneMode.Additive);
    try
    {
        var actors=scene.GetRootGameObjects().SelectMany(r=>r.GetComponentsInChildren<PlayerActor>(true)).ToArray();
        foreach(var actor in actors)
        {
            var player=actor.gameObject;
            var old=player.transform.Find("TechGuy_HD_Visual");
            if(!old) { if(player.transform.Find("TG_Rigged_Visual"))continue; throw new System.InvalidOperationException("Expected known player visual in "+path+" / "+player.name); }
            var animator=player.GetComponent<Animator>();
            if(!animator || !animator.avatar || (!animator.runtimeAnimatorController && path!="Assets/_Project/Scenes/ProceduralStageAuthoring.unity")) throw new System.InvalidOperationException("Configured root Animator is required: "+path);
            var oldAvatar=animator.avatar;
            var newVisual=(GameObject)PrefabUtility.InstantiatePrefab(model,scene);
            newVisual.name="TG_Rigged_Visual";
            newVisual.transform.SetParent(player.transform,false);
            newVisual.transform.localPosition=old.localPosition;
            newVisual.transform.localRotation=old.localRotation;
            newVisual.transform.localScale=old.localScale;
            var transforms=newVisual.GetComponentsInChildren<Transform>(true).ToDictionary(t=>t.name,t=>t);
            var objects=new System.Collections.Generic.Dictionary<UnityEngine.Object,UnityEngine.Object>();
            objects[old]=newVisual.transform; objects[old.gameObject]=newVisual;
            foreach(var bone in oldAvatar.humanDescription.human)
            {
                var oldTransform=old.GetComponentsInChildren<Transform>(true).SingleOrDefault(t=>t.name==bone.boneName);
                var newBone=avatar.humanDescription.human.SingleOrDefault(h=>h.humanName==bone.humanName);
                Transform target;
                if(oldTransform && !string.IsNullOrEmpty(newBone.boneName) && transforms.TryGetValue(newBone.boneName,out target)) { objects[oldTransform]=target; objects[oldTransform.gameObject]=target.gameObject; }
            }
            var oldRenderer=old.GetComponentsInChildren<SkinnedMeshRenderer>(true).Single();
            var newRenderer=newVisual.GetComponentsInChildren<SkinnedMeshRenderer>(true).Single();
            objects[oldRenderer]=newRenderer; objects[oldRenderer.transform]=newRenderer.transform; objects[oldRenderer.gameObject]=newRenderer.gameObject;
            var oldHips=oldAvatar.humanDescription.human.Single(h=>h.humanName=="Hips").boneName;
            var oldArmature=old.GetComponentsInChildren<Transform>(true).Single(t=>t.name==oldHips).parent;
            var newHips=transforms[avatar.humanDescription.human.Single(h=>h.humanName=="Hips").boneName];
            if(oldArmature!=old) { objects[oldArmature]=newHips.parent; objects[oldArmature.gameObject]=newHips.parent.gameObject; }
            var oldObjects=new System.Collections.Generic.HashSet<UnityEngine.Object>(old.GetComponentsInChildren<Component>(true).Cast<UnityEngine.Object>().Concat(old.GetComponentsInChildren<Transform>(true).Select(t=>(UnityEngine.Object)t.gameObject)));
            var changes=new System.Collections.Generic.List<string>();
            foreach(var component in scene.GetRootGameObjects().SelectMany(r=>r.GetComponentsInChildren<Component>(true)))
            {
                if(!component || component is Transform || component.transform.IsChildOf(old))continue;
                var serialized=new SerializedObject(component);
                var property=serialized.GetIterator();
                bool changed=false;
                while(property.Next(true))
                {
                    if(property.propertyType!=SerializedPropertyType.ObjectReference || !property.objectReferenceValue || !oldObjects.Contains(property.objectReferenceValue))continue;
                    UnityEngine.Object target;
                    if(!objects.TryGetValue(property.objectReferenceValue,out target)) throw new System.InvalidOperationException("Unmapped reference "+component.gameObject.name+" / "+component.GetType().Name+"."+property.propertyPath);
                    property.objectReferenceValue=target; changed=true;
                    changes.Add(component.gameObject.name+" / "+component.GetType().Name+"."+property.propertyPath);
                }
                if(changed) { serialized.ApplyModifiedPropertiesWithoutUndo(); PrefabUtility.RecordPrefabInstancePropertyModifications(component); }
            }
            foreach(var nestedAnimator in newVisual.GetComponentsInChildren<Animator>(true)) UnityEngine.Object.DestroyImmediate(nestedAnimator);
            animator.avatar=avatar;
            EditorUtility.SetDirty(animator);
            PrefabUtility.RecordPrefabInstancePropertyModifications(animator);
            UnityEngine.Object.DestroyImmediate(old.gameObject);
            animator.Rebind();
            if(!actor.HandTransform || !actor.HandTransform.IsChildOf(newVisual.transform)) throw new System.InvalidOperationException("Player weapon socket is not on the new rig.");
            var hand=animator.GetBoneTransform(HumanBodyBones.RightHand);
            if(hand!=actor.HandTransform) throw new System.InvalidOperationException("Player socket differs from the imported Humanoid right hand.");
            if(animator.applyRootMotion) throw new System.InvalidOperationException("Unexpected root-motion change.");
            results.Add(new { scene=path,player=player.name,avatar=avatar.name,hand=hand.name,remappedReferences=changes.ToArray(),controller=AssetDatabase.GetAssetPath(animator.runtimeAnimatorController) });
        }
        UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);
        if(!UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene)) throw new System.InvalidOperationException("Could not save "+path);
    }
    finally
    {
        if(opened) UnityEditor.SceneManagement.EditorSceneManager.CloseScene(scene,true);
    }
}
if(active.IsValid() && active.isLoaded) UnityEngine.SceneManagement.SceneManager.SetActiveScene(active);
return results.ToArray();
